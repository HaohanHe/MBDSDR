# §Title
# 深度思考

# §引言
# 深度思考让模型在生成最终回答前进行深度推理，通过内部思维链（Chain of Thought）逐步分析问题，
# 显著提升复杂任务的准确性。适用于复杂推理、代码生成、数学计算、多步骤分析等需要深度分析的场景。
#
# 核心能力：
# - 深度推理：将复杂问题拆解为多个步骤逐步分析，提升推理准确性
# - 思考过程透明：返回完整思考过程，增强可解释性
# - 灵活控制：支持开启/关闭，可根据任务复杂度按需使用

# §支持的模型
# 当前支持 mimo-v2.6-flash，mimo-v2.6-pro，mimo-v2.6-pro-ultraspeed，mimo-v2.5-pro 和 mimo-v2.5 模型。

# §请求参数
# 在请求中设置 thinking.type 参数控制深度思考开关：enabled 开启思考或 disabled 关闭思考。
# 默认开启：mimo-v2.6-flash，mimo-v2.6-pro，mimo-v2.6-pro-ultraspeed，mimo-v2.5-pro 和 mimo-v2.5

# §注意事项 / 参数限制
# 在深度思考下，上述模型不支持自定义 temperature 和 top_p 参数。即使传入该参数，
# 实际生效值也会被模型强制采用其推荐默认值 1.0 和 0.95。

# §注意事项 / 多轮对话回传要求（核心红线）
# 在 Agent 类产品的多轮会话中开启深度思考，且历史会话中存在工具调用时，后续所有 user 交互轮次中
# 回传的 assistant 如果包含了工具调用，必须完整回传 reasoning_content 字段，否则 API 将返回 400 错误。
# 正确回传方式请参考调用示例的"深度思考下的多轮工具调用"。
#
# 历史 reasoning_content 一旦缺失，模型上下文将不完整，可能表现出指令遵循下降，幻觉增多等现象。
#
# 受影响的 Agent 产品：
# | 协议 | 受影响的 Agent 产品 |
# | OpenAI 兼容协议 | TRAE、Cursor、Roo Code、Codex、GitHub Copilot CLI、Zed、AutoGen、Goose |
# | Anthropic 兼容协议 | TRAE、GitHub Copilot CLI、AutoGen、Goose、OpenClaw、OpenCode、Kilo Code |

# §注意事项 / 其他说明
# 1. 输出长度限制：max_completion_tokens 限制的是思考内容与最终回答的总长度。
#    若思考过程较长，留给最终回答的 token 空间会相应减少。建议设置足够的 max_completion_tokens 以避免回答被截断。
# 2. 响应时间：开启深度思考会增加响应延迟，复杂任务尤为明显。建议结合 stream: true 实时查看思考过程。

# §调用示例
# thinking 字段并非 OpenAI 标准参数。通过 OpenAI Python SDK 传入思考相关参数时，
# 需将其置于 extra_body 中传递。

# §开启思考（请求）
# client = OpenAI(api_key=os.environ.get("MIMO_API_KEY"), base_url="https://api.xiaomimimo.com/v1")
# completion = client.chat.completions.create(
#   model="mimo-v2.6-pro",
#   messages=[{role:system,...},{role:user,content:"Introduce machine learning in three sentences."}],
#   max_completion_tokens=1024,
#   extra_body={"thinking": {"type": "enabled"}}
# )

# §开启思考（响应示例，关键字段）
# choices[0].message = {
#   "content": "Machine learning is ...",
#   "role": "assistant",
#   "tool_calls": null,
#   "reasoning_content": "The user wants a concise three-sentence introduction to machine learning."
# }
# usage.completion_tokens_details.reasoning_tokens = 14
# usage.prompt_tokens_details.cached_tokens = 0
# —— 即：reasoning_content 是与 content/tool_calls 同级、挂在 message 上的字段；
#      开启思考时出现，关闭思考时该字段不出现（见关闭思考响应：message 无 reasoning_content，reasoning_tokens=0）。

# §关闭思考
# extra_body={"thinking": {"type": "disabled"}}；响应 message 无 reasoning_content 字段，
# completion_tokens_details.reasoning_tokens = 0。

# §流式响应（开启思考）
# 流式响应时，思考内容与回答内容依次输出：首先通过 reasoning_content 逐步返回思考过程，
# 思考完成后，再通过 content 逐步输出最终回答。
#
# chunk 结构：choices[0].delta = {"content":..., "role":..., "tool_calls":null, "reasoning_content":...}
# - 首 chunk：delta={"content":"","role":"assistant","tool_calls":null,"reasoning_content":null}
# - 思考阶段：delta.reasoning_content 逐段累加（如 "The user is asking" / " for tips to" / " improve work efficiency."），
#   此时 delta.content=null
# - 切换到回答：delta.content 开始出字（如 "# Tips" / " for Improving Work"），此时 delta.reasoning_content=null
# - 结束 chunk：delta 全 null，finish_reason="stop"
# - 最后一个 chunk：choices=[]，带 usage（completion_tokens_details.reasoning_tokens=29）
# - data: [DONE]
# —— 即：流式时 reasoning_content 走 delta.reasoning_content，必须与 delta.content 分别累加、不可混。

# §思考模式下的多轮工具调用（官方完整示例，逐字段记录）
# 引言：在深度思考的多轮对话中，若涉及工具调用，回传 reasoning_content 可确保思考连续性，提升模型输出质量。
#
# 工具定义（两个 function）：
#   get_current_weather(location:string[required], unit:enum[celsius,fahrenheit])
#   get_time(timezone:string[required])
#
# run_turn(messages, turn_num) 循环伪代码（官方原样逻辑）：
#   while True:
#     response = client.chat.completions.create(
#         model="mimo-v2.6-pro", messages=messages, tools=tools,
#         extra_body={"thinking": {"type": "enabled"}})
#     assistant_message = response.choices[0].message
#     messages.append(assistant_message)          # ← 关键：整条 assistant_message（含 reasoning_content/content/tool_calls）原样 append
#     if not assistant_message.tool_calls:
#         break                                   # 无工具调用 = 最终回答，退出循环
#     for tool_call in assistant_message.tool_calls:
#         func_name = tool_call.function.name
#         func_args = json.loads(tool_call.function.arguments)
#         result = TOOL_MAP[func_name](**func_args)
#         messages.append({                       # ← tool 结果以 role:"tool" 回传
#             "role": "tool",
#             "tool_call_id": tool_call.id,
#             "content": result
#         })
#
# 多轮会话：
#   Turn 1: messages.append({role:user, content:"How is the weather in Beijing today? What time is it now?"})
#           run_turn(messages, turn_num=1)
#   Turn 2: messages.append({role:user, content:"How about Shanghai? And is it hotter or colder than Beijing?"})
#           run_turn(messages, turn_num=2)
#   —— 注释原文：Turn 2 时 "reasoning_content from Turn 1 is already in messages via assistant_message"
#
# 示例输出（Request 1-1，模型决定并行调两个工具）：
#   reasoning_content: "The user wants both weather and time in Beijing. I can call both in parallel.
#                       Weather: get_current_weather(location: Beijing). Time: Beijing timezone is Asia/Shanghai.
#                       Call both together."
#   content: ""
#   tool_calls: [
#     {id:'call_01e402113df94ebfb85e3056', function:{arguments:'{"location": "Beijing"}', name:'get_current_weather'}, type:'function'},
#     {id:'call_f2bf1ab9f73842baa64e93d3', function:{arguments:'{"timezone": "Asia/Shanghai"}', name:'get_time'}, type:'function'}]
#   -> Tool result [get_current_weather]: Sunny 25°C
#   -> Tool result [get_time]: 2026-09-22 00:38:45 (Asia/Shanghai)
#
# 示例输出（Request 1-2，工具结果后模型生成最终回答）：
#   reasoning_content: "Present both results clearly."
#   content: "Here's the current information for Beijing: ..."
#   tool_calls: None
#
# 示例输出（Request 2-1，第二轮 user 后模型再次调工具）：
#   reasoning_content: "Need weather and time for Shanghai, and compare to Beijing's 25°C. ..."
#   content: ""
#   tool_calls: [get_current_weather(Shanghai), get_time(Asia/Shanghai)]
#
# 示例输出（Request 2-2）：
#   reasoning_content: "Shanghai: cloudy, 22°C vs Beijing sunny 25°C. Shanghai is 3°C colder. ..."
#   content: "Here's the current information for Shanghai: ... Comparison: Shanghai is colder than Beijing ..."
#   tool_calls: None
#
# 官方说明原文：由于第一轮的 assistant 消息（包含 reasoning_content、content、tool_calls）已通过
# messages.append() 累积到对话历史中，模型能直接从上下文获取北京 25°C 的信息。
